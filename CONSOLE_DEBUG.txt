// Copy và paste toàn bộ đoạn này vào Browser Console (F12)

(function() {
    console.log('=== TOM SELECT DEBUG ===');
    console.log('');
    
    // 1. Check JobForm loaded
    console.log('1. window.JobForm:', typeof window.JobForm);
    if (window.JobForm) {
        console.log('   Methods:', Object.keys(window.JobForm));
    }
    console.log('');
    
    // 2. Check elements
    const skillsEl = document.querySelector('#required_skills');
    const benefitsEl = document.querySelector('#job_benefits');
    console.log('2. Elements exist?');
    console.log('   Skills element:', skillsEl ? '✅' : '❌');
    console.log('   Benefits element:', benefitsEl ? '✅' : '❌');
    console.log('');
    
    // 3. Check Tom Select instances
    console.log('3. Tom Select initialized?');
    console.log('   Skills tomselect:', skillsEl?.tomselect ? '✅' : '❌');
    console.log('   Benefits tomselect:', benefitsEl?.tomselect ? '✅' : '❌');
    console.log('');
    
    // 4. Check options count
    if (skillsEl?.tomselect) {
        const optCount = Object.keys(skillsEl.tomselect.options).length;
        console.log('4. Skills options count:', optCount);
        if (optCount === 0) {
            console.log('   ⚠️ NO OPTIONS! This is the problem.');
        } else {
            console.log('   ✅ Has options. First 3:', Object.values(skillsEl.tomselect.options).slice(0, 3));
        }
    } else {
        console.log('4. ❌ Cannot check options - Tom Select not initialized');
    }
    console.log('');
    
    // 5. Test API
    console.log('5. Testing API...');
    fetch('/api/jobs/options/form-data')
        .then(r => r.json())
        .then(data => {
            const skills = data.data?.attributes?.required_skills?.options;
            const benefits = data.data?.attributes?.job_benefits?.options;
            console.log('   API Skills:', skills ? `✅ ${skills.length} items` : '❌ No data');
            console.log('   API Benefits:', benefits ? `✅ ${benefits.length} items` : '❌ No data');
            
            if (skills && skills.length > 0) {
                console.log('   First 3 skills:', skills.slice(0, 3));
            }
        })
        .catch(e => console.error('   API Error:', e));
    
    console.log('');
    console.log('=== END DEBUG ===');
    console.log('');
    console.log('➡️ Paste the output above to help diagnose the issue');
})();
